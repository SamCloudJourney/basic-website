                selectorUrlAuthority = request.Url?.Authority ?? string.Empty;

                return selected;
            };

            Task<HttpListenerContext> adminTask = adminListener.GetContextAsync();

                "\r\n";

            context.Response.Close();

            Assert.False(publicTask.IsCompletedSuccessfully);
            Assert.False(adminTask.IsCompletedSuccessfully);
